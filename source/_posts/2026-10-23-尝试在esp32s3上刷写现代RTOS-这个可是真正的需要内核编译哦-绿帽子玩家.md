<pre><code class="language-bash">docker pull xrimage/xrimage-esp32

docker run -it \
	-v "$HOME/WORKS/esp32:/root/workspace" \
	-w /root/workspace \
	--network host \
	--privileged \
	-e DISPLAY="$DISPLAY" \
	--name esp32-dev \
	xrimage/xrimage-esp32 bash
	
# 等待读条...


#(root@bi7dyv:~/workspace#) docker中 (之后 docker exec -it esp32-dev bash )
cd /root/esp/esp-idf/
./install.sh
. ./export.sh

git clone --depth=1 git@github.com:apache/nuttx.git nuttx &amp;&amp; git clone --depth=1 git@github.com:apache/nuttx-apps.git apps
cd /root/nuttx/tools # ~/WORKS/esp32/nuttx/tools
apt install kconfig-frontends

./configure.sh esp32s3-devkitc:nsh # https://nuttx.apache.org/docs/latest/platforms/xtensa/esp32s3/boards/esp32s3-devkit/index.html
cd /root/nuttx
make menuconfig
# ./configure.sh esp32-devkitc:nsh 默认FLAT构建；有NuttX libc；
# 默认不开启uClibC++ (还有与其相关的"C++ 基础支持", "LLVM libc++" (?)也没开启); 默认不开启网络栈和wifi; 
# 注意！！NuttX使用它自己的 "NuttX libc" , 我们推荐使用uClibC++/LLVM libc++作为上层C++标准库
# 文件系统还不懂(有procfs 能以文件形式访问设备等等(?), 推荐用SPI Flash + LittleFS)
# (好像/dev完全不在RAM里！操作系统内核遍历任务管理链表, 临时凑出这些 inode)
# 
# 可能要注意一下stack的分配emm,这一块不懂
# 好吧我还要看看Kconfig和make menuconfig相关的东西 (似乎这个RTOS不怎么依赖设备树)
# https://www.cnblogs.com/chorm590/p/13977818.html
# 完蛋好像esp32s3也没有mpu不能进行隔离。。。std::thread也只相当于普通的TCB 。。。
